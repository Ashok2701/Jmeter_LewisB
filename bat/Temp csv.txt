DECLARE @LOGIN VARCHAR(50) = 'TLINDAUER'
DECLARE @PASSWORD VARCHAR(50) = '1234'

SELECT 
    D.*, 
    D.DRIVERID_0 AS Driver, 
    D.DRIVER_0 AS Description, 
    D.ACTIVE_0 AS Active, 
    D.XSALEREP_0 AS Sales_Rep, 
    F.FCY_0 AS Site, 
    F.FCYNAM_0 AS Site_Description, 
    A.XFLG_0 AS XFLG 
FROM sagex3.PILOT.XX10CDRIVER D 
LEFT JOIN sagex3.PILOT.FACILITY F 
    ON D.FCY_0 = F.FCY_0 
OUTER APPLY 
( 
    SELECT TOP 1 M.XFLG_0 
    FROM sagex3.PILOT.X10CADMVP M 
    WHERE UPPER(LTRIM(RTRIM(M.XSALADMIN_0))) 
        = UPPER(LTRIM(RTRIM(D.DRIVERID_0))) 
    ORDER BY M.XLINENO_0 
) A 
WHERE UPPER(LTRIM(RTRIM(D.XUSER_0))) 
        = UPPER(LTRIM(RTRIM(@LOGIN))) 
  AND D.XPWD1_0 = @PASSWORD 
  AND D.ACTIVE_0 = 2;
